using Aspose.Cells;
using FluentAssertions;
using FluentValidation;
using FluentValidation.Results;
using Hero.Integration.Aspose;
using Hero.Integration.Aspose.MultipleProperties;
using Hero.Integration.CoreApi;
using Hero.Integration.CoreApi.Interfaces;
using Hero.Integration.Geolocation;
using Hero.Integration.PriorSubmit;
using Hero.Models;
using Hero.Models.Validation;
using Microsoft.Extensions.Logging;
using NSubstitute;
using NUnit.Framework;
using WebApiDto.Dto;
using ValidationResult = Hero.Models.Validation.ValidationResult;
using PropertyLimit = Hero.Models.PropertyLimit;

namespace Hero.Tests.PropertyTemplate;

[TestFixture]
public class BaseMultiplePropertyOrchestratorTests
{
    private ILogger<TestMultiplePropertyOrchestrator> _logger;
    private ILocationApi _locationApi;
    private IAsposeRowExtractor _rowExtractor;
    private IAsposeRowInserter _rowInserter;
    private IGeolocationService _geolocationService;
    private IValidator<LocationValidationRequest> _locationValidator;
    private IListOfCountries _countries;
    private ITemplateColumnValidator _templateColumnValidator;
    private IPriorSubmitApprovalService _priorSubmitApprovalService;

    private TestMultiplePropertyOrchestrator _subject;
    private Workbook _workbook;

    [SetUp]
    public void SetUp()
    {
        _logger = Substitute.For<ILogger<TestMultiplePropertyOrchestrator>>();
        _locationApi = Substitute.For<ILocationApi>();
        _rowExtractor = Substitute.For<IAsposeRowExtractor>();
        _rowInserter = Substitute.For<IAsposeRowInserter>();
        _geolocationService = Substitute.For<IGeolocationService>();
        _locationValidator = Substitute.For<IValidator<LocationValidationRequest>>();
        _countries = Substitute.For<IListOfCountries>();
        _templateColumnValidator = Substitute.For<ITemplateColumnValidator>();
        _priorSubmitApprovalService = Substitute.For<IPriorSubmitApprovalService>();

        _workbook = new Workbook();

        _subject = new TestMultiplePropertyOrchestrator(
            _logger,
            _locationApi,
            _rowExtractor,
            _rowInserter,
            _geolocationService,
            _locationValidator,
            _countries,
            _templateColumnValidator,
            _priorSubmitApprovalService);

        _locationValidator
            .ValidateAsync(Arg.Any<LocationValidationRequest>())
            .Returns(new FluentValidation.Results.ValidationResult());

        _priorSubmitApprovalService
            .EvaluateLocationsAsync(Arg.Any<IList<PropertyLimit>>())
            .Returns(Task.CompletedTask);
    }

    [TearDown]
    public void TearDown()
    {
        _workbook.Dispose();
    }

    [Test]
    public async Task GivenTemplateValidationFails_WhenIGetPropertyLimits_ThenTemplateValidationErrorIsReturned()
    {
        // Arrange
        _subject.TemplateValidationResult = (false, "Bad template");

        // Act
        var result = await _subject.GetPropertyLimitsAsync(_workbook, 123, 1);

        // Assert
        result.TemplateValidationError.Should().Be("Bad template");
        result.PropertyLimits.Should().BeNull();
        result.ValidationResults.Should().BeNull();

        _rowExtractor.DidNotReceive()
            .ExtractRows(Arg.Any<Workbook>(), Arg.Any<ExtractorColumnConfiguration>(), Arg.Any<int>(), Arg.Any<bool>());

        await _priorSubmitApprovalService.DidNotReceive()
            .EvaluateLocationsAsync(Arg.Any<IList<PropertyLimit>>());
    }

    [Test]
    public async Task GivenTemplateValidationSucceeds_WhenIGetPropertyLimits_ThenReturnsBuiltUploadResult()
    {
        // Arrange
        _subject.TemplateValidationResult = (true, string.Empty);
        _subject.LocationRows = new List<ExtractedRow>
        {
            TestMultiplePropertyOrchestrator.CreateRow("10 High Street", "SW1A 1AA", 2)
        };
        _subject.ContextToReturn = new TestUploadContext();
        _subject.PropertyLimitsToReturn = new List<PropertyLimit>
        {
            new() { RowNumber = 2 }
        };

        // Act
        var result = await _subject.GetPropertyLimitsAsync(_workbook, 123, 1);

        // Assert
        result.TemplateValidationError.Should().BeNull();
        result.PropertyLimits.Should().HaveCount(1);
        result.ValidationResults.Should().NotBeNull();
    }

    [Test]
    public async Task GivenTemplateValidationSucceeds_WhenIGetPropertyLimits_ThenPriorSubmitIsCalledWithBuiltPropertyLimits()
    {
        // Arrange
        _subject.TemplateValidationResult = (true, string.Empty);
        _subject.LocationRows = new List<ExtractedRow>
        {
            TestMultiplePropertyOrchestrator.CreateRow("10 High Street", "SW1A 1AA", 2)
        };
        _subject.PropertyLimitsToReturn = new List<PropertyLimit>
        {
            new() { RowNumber = 2 }
        };

        // Act
        await _subject.GetPropertyLimitsAsync(_workbook, 123, 1);

        // Assert
        await _priorSubmitApprovalService.Received(1)
            .EvaluateLocationsAsync(Arg.Is<IList<PropertyLimit>>(x => x.Count == 1 && x[0].RowNumber == 2));
    }

    [Test]
    public async Task GivenLocationValidationFails_WhenHasInvalidLocationValidationAsyncIsCalled_ThenValidationResultIsAdded()
    {
        // Arrange
        var location = new LocationValidationRequest
        {
            Address1 = "1 Test Street",
            City = "London",
            CountryName = "UK",
            Postcode = "SW1A 1AA"
        };
        var row = new ExtractedRow { RowNumber = 9 };
        var validationResults = new List<ValidationResult>();

        _locationValidator
            .ValidateAsync(location)
            .Returns(new FluentValidation.Results.ValidationResult(new[]
            {
                new ValidationFailure("Address1", "Address is invalid")
            }));

        // Act
        var invalid = await _subject.CallHasInvalidLocationValidationAsync(validationResults, location, row);

        // Assert
        invalid.Should().BeTrue();
        validationResults.Should().ContainSingle();
        validationResults[0].LocationIdentifier.Should().Be("Row 9");
        validationResults[0].Errors.Should().ContainSingle();
        validationResults[0].Errors[0].ErrorMessage.Should().Be("Address is invalid");
    }

    [Test]
    public async Task GivenLocationValidationPasses_WhenHasInvalidLocationValidationAsyncIsCalled_ThenNoValidationResultIsAdded()
    {
        // Arrange
        var location = new LocationValidationRequest
        {
            Address1 = "1 Test Street",
            City = "London",
            CountryName = "UK",
            Postcode = "SW1A 1AA"
        };
        var row = new ExtractedRow { RowNumber = 9 };
        var validationResults = new List<ValidationResult>();

        _locationValidator
            .ValidateAsync(location)
            .Returns(new FluentValidation.Results.ValidationResult());

        // Act
        var invalid = await _subject.CallHasInvalidLocationValidationAsync(validationResults, location, row);

        // Assert
        invalid.Should().BeFalse();
        validationResults.Should().BeEmpty();
    }

    [Test]
    public void GivenDuplicateAddresses_WhenIdentifyDuplicateAddressKeysIsCalled_ThenDuplicateKeyIsReturned()
    {
        // Arrange
        var rows = new List<ExtractedRow>
        {
            TestMultiplePropertyOrchestrator.CreateRow("10 High Street", "SW1A 1AA", 2),
            TestMultiplePropertyOrchestrator.CreateRow("10 High Street", "SW1A 1AA", 3),
            TestMultiplePropertyOrchestrator.CreateRow("20 Main Road", "M1 1AA", 4)
        };

        // Act
        var result = TestMultiplePropertyOrchestrator.CallIdentifyDuplicateAddressKeys(rows);

        // Assert
        result.Should().ContainSingle();
        result.Should().Contain("10 HIGH STREET|SW1A 1AA");
    }

    [Test]
    public void GivenNonDuplicateAddresses_WhenIdentifyDuplicateAddressKeysIsCalled_ThenEmptySetIsReturned()
    {
        // Arrange
        var rows = new List<ExtractedRow>
        {
            TestMultiplePropertyOrchestrator.CreateRow("10 High Street", "SW1A 1AA", 2),
            TestMultiplePropertyOrchestrator.CreateRow("20 Main Road", "M1 1AA", 3)
        };

        // Act
        var result = TestMultiplePropertyOrchestrator.CallIdentifyDuplicateAddressKeys(rows);

        // Assert
        result.Should().BeEmpty();
    }

    [Test]
    public void GivenAddressIsDuplicate_WhenHasDuplicateAddressIsCalled_ThenValidationResultIsAdded()
    {
        // Arrange
        var duplicateKeys = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
        {
            "10 HIGH STREET|SW1A 1AA"
        };

        var validationResults = new List<ValidationResult>();
        var location = new LocationValidationRequest
        {
            Address1 = "10 High Street",
            Postcode = "SW1A 1AA",
            City = "London",
            CountryName = "UK"
        };
        var row = new ExtractedRow { RowNumber = 4 };

        // Act
        var isDuplicate = TestMultiplePropertyOrchestrator.CallHasDuplicateAddress(
            validationResults,
            location,
            duplicateKeys,
            row);

        // Assert
        isDuplicate.Should().BeTrue();
        validationResults.Should().ContainSingle();
        validationResults[0].LocationIdentifier.Should().Be("Row 4");
        validationResults[0].Errors[0].ErrorMessage.Should().Contain("Duplicate address found");
    }

    [Test]
    public void GivenAddressIsNotDuplicate_WhenHasDuplicateAddressIsCalled_ThenNoValidationResultIsAdded()
    {
        // Arrange
        var duplicateKeys = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
        {
            "10 HIGH STREET|SW1A 1AA"
        };

        var validationResults = new List<ValidationResult>();
        var location = new LocationValidationRequest
        {
            Address1 = "20 Main Road",
            Postcode = "M1 1AA",
            City = "Manchester",
            CountryName = "UK"
        };
        var row = new ExtractedRow { RowNumber = 4 };

        // Act
        var isDuplicate = TestMultiplePropertyOrchestrator.CallHasDuplicateAddress(
            validationResults,
            location,
            duplicateKeys,
            row);

        // Assert
        isDuplicate.Should().BeFalse();
        validationResults.Should().BeEmpty();
    }
}

using Aspose.Cells;
using FluentValidation;
using Hero.Integration.Aspose;
using Hero.Integration.Aspose.MultipleProperties;
using Hero.Integration.CoreApi;
using Hero.Integration.CoreApi.Interfaces;
using Hero.Integration.Geolocation;
using Hero.Integration.PriorSubmit;
using Hero.Models;
using Hero.Models.Validation;
using Microsoft.Extensions.Logging;
using ValidationResult = Hero.Models.Validation.ValidationResult;
using PropertyLimit = Hero.Models.PropertyLimit;
using PropertyLimitFloatingValues = WebApiDto.Dto.PropertyLimitFloatingValues;

namespace Hero.Tests.PropertyTemplate;

internal sealed class TestMultiplePropertyOrchestrator : BaseMultiplePropertyOrchestrator<TestUploadContext>
{
    public (bool isValid, string error) TemplateValidationResult { get; set; } = (true, string.Empty);
    public List<ExtractedRow> LocationRows { get; set; } = new();
    public TestUploadContext ContextToReturn { get; set; } = new();
    public List<PropertyLimit> PropertyLimitsToReturn { get; set; } = new();

    public TestMultiplePropertyOrchestrator(
        ILogger logger,
        ILocationApi locationApi,
        IAsposeRowExtractor asposeRowExtractor,
        IAsposeRowInserter asposeRowInserter,
        IGeolocationService geolocationService,
        IValidator<LocationValidationRequest> locationValidator,
        IListOfCountries countries,
        ITemplateColumnValidator templateColumnValidator,
        IPriorSubmitApprovalService priorSubmitApprovalService)
        : base(
            logger,
            locationApi,
            asposeRowExtractor,
            asposeRowInserter,
            geolocationService,
            locationValidator,
            countries,
            templateColumnValidator,
            priorSubmitApprovalService)
    {
    }

    protected override (bool isValid, string error) ValidateTemplateColumns(Workbook workbook, int wordingVersionId)
        => TemplateValidationResult;

    protected override List<ExtractedRow> ExtractLocationRows(Workbook workbook, int wordingVersionId)
        => LocationRows;

    protected override Task<TestUploadContext> ExtractContextAsync(
        Workbook workbook,
        List<ValidationResult> validationResults)
        => Task.FromResult(ContextToReturn);

    protected override Task<List<PropertyLimit>> BuildPropertyLimitsAsync(
        List<ExtractedRow> locationRows,
        TestUploadContext context,
        List<ValidationResult> validationResults)
        => Task.FromResult(PropertyLimitsToReturn);

    protected override TemplateUploadResult BuildUploadResult(
        List<PropertyLimit> propertyLimits,
        TestUploadContext context,
        List<ValidationResult> validationResults)
    {
        return new TemplateUploadResult
        {
            PropertyLimits = propertyLimits,
            ValidationResults = validationResults
        };
    }

    public override Task<Workbook> InsertPropertyLimitsIntoWorkbook(
        Workbook workbook,
        IEnumerable<WebApiDto.Dto.PropertyLimit> propertyLimits,
        PropertyLimitFloatingValues propertyLimitFloatingValues,
        int? firstLossLimit,
        int wordingVersionId)
    {
        return Task.FromResult(workbook);
    }

    public Task<bool> CallHasInvalidLocationValidationAsync(
        List<ValidationResult> validationResults,
        LocationValidationRequest location,
        ExtractedRow row)
        => HasInvalidLocationValidationAsync(validationResults, location, row);

    public static HashSet<string> CallIdentifyDuplicateAddressKeys(List<ExtractedRow> rows)
        => IdentifyDuplicateAddressKeys(rows, CreateLocationRequestFrom);

    public static bool CallHasDuplicateAddress(
        List<ValidationResult> validationResults,
        LocationValidationRequest location,
        HashSet<string> duplicateKeys,
        ExtractedRow row)
        => HasDuplicateAddress(validationResults, location, duplicateKeys, row);

    private static LocationValidationRequest CreateLocationRequestFrom(ExtractedRow row)
    {
        return new LocationValidationRequest
        {
            Address1 = row["AddressLine1"]?.StringValue,
            Postcode = row["PostCode"]?.StringValue,
            City = row["TownOrCity"]?.StringValue,
            CountryName = row["Country"]?.StringValue
        };
    }

    public static ExtractedRow CreateRow(string address1, string postcode, int rowNumber)
    {
        return new ExtractedRow
        {
            RowNumber = rowNumber,
            CellValues = new List<ExtractedCellValue>
            {
                new() { ColumnName = "AddressLine1", StringValue = address1 },
                new() { ColumnName = "PostCode", StringValue = postcode },
                new() { ColumnName = "TownOrCity", StringValue = "London" },
                new() { ColumnName = "Country", StringValue = "UK" }
            }
        };
    }
}

internal sealed class TestUploadContext
{
}
